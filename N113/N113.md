

import pandas as pd

import matplotlib.pyplot as plt

import seaborn as sns



from sklearn.model\_selection import train\_test\_split

from sklearn.preprocessing import StandardScaler

from sklearn.linear\_model import LogisticRegression

from sklearn.tree import DecisionTreeClassifier

from sklearn.ensemble import RandomForestClassifier

from sklearn.metrics import accuracy\_score, confusion\_matrix, precision\_recall\_fscore\_support





TARGET\_COLUMN = "Kết quả bệnh tim"

FEATURE\_COLUMNS = \["Tuổi", "Giới tính", "Huyết áp lúc nghỉ", "Cholesterol", "Nhịp tim tối đa"]





print(">>> Bắt đầu chương trình...\\n")



data = pd.read\_excel("heart.xlsx")



print("===== BƯỚC 1: XEM QUA DỮ LIỆU =====")

print("5 dòng đầu tiên trong file:")

print(data.head())



print("\\nTổng số người có bệnh và không bệnh:")

print(data\[TARGET\_COLUMN].value\_counts())





print("\\n===== BƯỚC 2: VẼ BIỂU ĐỒ =====")



plt.figure(figsize=(5, 4))

sns.countplot(x=TARGET\_COLUMN, data=data)

plt.title("So sánh số người có bệnh và không bệnh")

plt.savefig("bieu\_do\_1\_phan\_bo.png")

plt.close()

print("Đã vẽ xong: bieu\_do\_1\_phan\_bo.png")



plt.figure(figsize=(8, 6))

sns.heatmap(data.corr(numeric\_only=True), annot=True, cmap="coolwarm")

plt.title("Mức độ liên quan giữa các chỉ số")

plt.savefig("bieu\_do\_2\_tuong\_quan.png")

plt.close()

print("Đã vẽ xong: bieu\_do\_2\_tuong\_quan.png")





print("\\n===== BƯỚC 3: CHUẨN BỊ DỮ LIỆU =====")



X = data\[FEATURE\_COLUMNS]

y = data\[TARGET\_COLUMN]



X\_train, X\_test, y\_train, y\_test = train\_test\_split(

&#x20;   X, y, test\_size=0.2, random\_state=42

)

print(f"Số dòng dùng để học: {len(X\_train)}")

print(f"Số dòng dùng để kiểm tra: {len(X\_test)}")



scaler = StandardScaler()

X\_train\_scaled = scaler.fit\_transform(X\_train)

X\_test\_scaled = scaler.transform(X\_test)





print("\\n===== BƯỚC 4: CHO MÁY HỌC =====")



model\_logistic = LogisticRegression()

model\_logistic.fit(X\_train\_scaled, y\_train)

print("Đã học xong: Hồi quy Logistic (Logistic Regression)")



model\_tree = DecisionTreeClassifier(random\_state=42)

model\_tree.fit(X\_train\_scaled, y\_train)

print("Đã học xong: Cây quyết định (Decision Tree)")



model\_forest = RandomForestClassifier(random\_state=42)

model\_forest.fit(X\_train\_scaled, y\_train)

print("Đã học xong: Rừng ngẫu nhiên (Random Forest)")







print("\\n===== BƯỚC 5: CHẤM ĐIỂM TỪNG MÔ HÌNH =====")



models = \[

&#x20;   ("Hồi quy Logistic", model\_logistic),

&#x20;   ("Cây quyết định", model\_tree),

&#x20;   ("Rừng ngẫu nhiên", model\_forest),

]



for model\_name, model in models:

&#x20;   predictions = model.predict(X\_test\_scaled)

&#x20;   accuracy = accuracy\_score(y\_test, predictions)



&#x20;   print(f"\\n--- {model\_name} ---")

&#x20;   print(f"Độ chính xác chung: {accuracy \* 100:.1f}%")



&#x20;   do\_chuan\_xac, do\_bao\_phu, diem\_f1, so\_luong = precision\_recall\_fscore\_support(

&#x20;       y\_test, predictions, zero\_division=0

&#x20;   )



&#x20;   print("\\nBáo cáo chi tiết theo từng nhóm:")

&#x20;   print(f"{'Nhóm':<18}{'Độ chuẩn xác':<15}{'Độ bao phủ':<15}{'Điểm F1':<12}{'Số lượng':<10}")

&#x20;   ten\_nhom = {0: "Không bệnh (0)", 1: "Có bệnh (1)"}

&#x20;   for i in range(len(do\_chuan\_xac)):

&#x20;       print(f"{ten\_nhom.get(i, i):<18}{do\_chuan\_xac\[i]:<15.2f}{do\_bao\_phu\[i]:<15.2f}{diem\_f1\[i]:<12.2f}{so\_luong\[i]:<10}")



&#x20;   print("\\nMa trận nhầm lẫn (hàng = thực tế, cột = máy đoán):")

&#x20;   print(confusion\_matrix(y\_test, predictions))
