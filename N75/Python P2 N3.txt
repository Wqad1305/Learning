class Solution(object):
    def longestBalanced(self, s):
        n=len(s)
        ans=0
        for i in range(n):
            for j in range(i,n):
                sub=s[i:j+1]
                count_a=0
                count_b=0
                count_c=0
                for ch in sub:
                    if ch =='a':
                        count_a+=1
                    elif ch=='b':
                        count_b+=1
                    else:
                        count_c+=1
                counts = []

                if count_a > 0:
                    counts.append(count_a)

                if count_b > 0:
                    counts.append(count_b)

                if count_c > 0:
                    counts.append(count_c)

                if len(set(counts)) == 1:
                    ans = max(ans, len(sub))

        return ans

class Solution(object):
    def findMax(self, nums):
        maximum = nums[0]

        for num in nums:
            if num > maximum:
                maximum = num

        return maximum

class Solution(object):
    def mostFrequent(self, nums):
        freq = {}

        for num in nums:
            freq[num] = freq.get(num, 0) + 1

        ans = nums[0]

        for num in freq:
            if freq[num] > freq[ans]:
                ans = num

        return ans

class Solution(object):
    def reverseArray(self, nums):
        left = 0
        right = len(nums) - 1

        while left < right:
            nums[left], nums[right] = nums[right], nums[left]
            left += 1
            right -= 1

        return nums

class Solution(object):
    def firstUnique(self, nums):
        count = {}

        for num in nums:
            count[num] = count.get(num, 0) + 1

        for num in nums:
            if count[num] == 1:
                return num

        return -1
