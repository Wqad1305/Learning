class Solution {

private:

&#x20;   struct Node {

&#x20;       Node\* child\[26];

&#x20;       bool isEnd;



&#x20;       Node() {

&#x20;           isEnd = false;



&#x20;           for (int i = 0; i < 26; i++) {

&#x20;               child\[i] = nullptr;

&#x20;           }

&#x20;       }

&#x20;   };



public:

&#x20;   string longestCommonPrefix(vector<string>\& strs) {

&#x20;       if (strs.empty()) {

&#x20;           return "";

&#x20;       }



&#x20;       Node\* root = new Node();



&#x20;       for (string\& word : strs) {

&#x20;           Node\* cur = root;



&#x20;           for (char c : word) {

&#x20;               int idx = c - 'a';



&#x20;               if (cur->child\[idx] == nullptr) {

&#x20;                   cur->child\[idx] = new Node();

&#x20;               }



&#x20;               cur = cur->child\[idx];

&#x20;           }



&#x20;           cur->isEnd = true;

&#x20;       }

&#x20;       string ans = "";

&#x20;       Node\* cur = root;



&#x20;       while (true) {

&#x20;           int count = 0;

&#x20;           int nextIndex = -1;



&#x20;           for (int i = 0; i < 26; i++) {

&#x20;               if (cur->child\[i] != nullptr) {

&#x20;                   count++;

&#x20;                   nextIndex = i;

&#x20;               }

&#x20;           }



&#x20;           if (count != 1 || cur->isEnd) {

&#x20;               break;

&#x20;           }



&#x20;           ans += char('a' + nextIndex);

&#x20;           cur = cur->child\[nextIndex];

&#x20;       }



&#x20;       return ans;

&#x20;   }

};

\----------------------------------------------------------



1. Xây Trie từ tất cả các từ trong strs.

2. Duyệt Trie từ root và đếm số child của mỗi node.

3. Nếu chỉ có 1 child và chưa gặp isEnd thì thêm ký tự vào kết quả.

4. Nếu có nhiều hơn 1 child hoặc gặp isEnd thì dừng.

5. Trả về ans là prefix chung dài nhất.
